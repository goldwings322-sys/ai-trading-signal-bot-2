function getSignal(emaFast, emaSlow, rsi, bars) { if ( emaFast === null || emaSlow === null || rsi === null || !bars || bars.length < 2 ) { return "WAIT"; } const last = bars[bars.length - 1]; const previous = bars[bars.length - 2]; const bullishCandle = last.close > last.open; const bearishCandle = last.close < last.open; // Flexible BUY: // Uptrend with bullish candle, or recovery from oversold RSI. const buyTrend = emaFast > emaSlow && rsi > 30 && rsi < 70 && bullishCandle; const buyRecovery = rsi > 20 && rsi < 35 && rsi > calculateRSI( bars.slice(0, -1).map(bar => bar.close), 14 ) && bullishCandle && last.close > previous.close; // Flexible SELL: // Downtrend with bearish candle, or rejection from overbought RSI. const sellTrend = emaFast < emaSlow && rsi > 40 && rsi < 65 && bearishCandle; const previousRSI = calculateRSI( bars.slice(0, -1).map(bar => bar.close), 14 ); const sellReversal = rsi < 80 && rsi > 65 && previousRSI !== null && rsi < previousRSI && bearishCandle && last.close < previous.close; if (buyRecovery) return "BUY"; if (sellReversal) return "SELL"; if (buyTrend) return "BUY"; if (sellTrend) return "SELL"; return "WAIT"; }
